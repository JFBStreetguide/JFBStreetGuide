    function saveStreetsToStorage(data) {
      data.sort((a, b) => (a.name || '').localeCompare(b.name || ''));

    function renderList() {
      const query = document.getElementById('searchInput').value.toLowerCase();
      let streets = getStreets();

      streets.sort((a, b) => (a.name || '').localeCompare(b.name || ''));

      const container = document.getElementById('streetContainer');
      container.innerHTML = '';

      const filtered = streets.filter(s => 
        (s.name && s.name.toLowerCase().includes(query)) ||
        (s.district && s.district.toLowerCase().includes(query)) ||
        (s.primary_station && s.primary_station.toLowerCase().includes(query))
      );

      document.getElementById('recordCount').innerText = `${filtered.length} of ${streets.length} Streets`;

      filtered.forEach(s => {
        const item = document.createElement('div');
        item.className = 'street-item';
        item.onclick = () => showDetail(s.id);
        item.innerHTML = `
          <div class="street-main">
            <div class="street-title">${s.name || 'UNKNOWN STREET'}</div>
            <div class="street-district">${s.district || 'NO DISTRICT'}</div>
          </div>
          <div class="station-badge">${s.primary_station || 'N/A'}</div>
        `;
        container.appendChild(item);
      });
    }

    function showDetail(id) {

      document.getElementById('d_primary').innerText = s.primary_station || '';
      document.getElementById('d_next').innerText = s.next_station || '';
      document.getElementById('d_remarks').innerText = s.remarks || 'NONE';

      openModal('detailModal');
    }

    /* AI Camera OCR Scanner */
    async function handleCardScan(e) {
      const file = e.target.files[0];
      if(!file) return;

      const preview = document.getElementById('scanPreview');
      const status = document.getElementById('scanStatus');
      
      preview.src = URL.createObjectURL(file);
      preview.style.display = 'block';
      status.innerText = 'Analyzing card with AI OCR... Please wait...';

      try {
        const worker = await Tesseract.createWorker('eng');
        const ret = await worker.recognize(file);
        await worker.terminate();

        const text = ret.data.text;
        status.innerText = 'Analysis complete! Review details below:';

        const lines = text.split('\n').map(l => l.trim()).filter(l => l.length > 0);

        lines.forEach(line => {
          const upper = line.toUpperCase();
          if(upper.includes('STREET') || upper.includes('AVENUE') || upper.includes('ROAD') || upper.includes('DRIVE') || upper.includes('LANE') || upper.includes('PLACE') || upper.includes('CRESCENT')) {
            if(!document.getElementById('s_name').value) document.getElementById('s_name').value = upper.replace('NAME OF STREET', '').trim();
          }
          if(upper.includes('HALF WAY TREE') || upper.includes('TRENCH TOWN') || upper.includes('HEADQUARTERS') || upper.includes('ROLLINGTON TOWN') || upper.includes('STONY HILL')) {
            if(!document.getElementById('s_primary').value) document.getElementById('s_primary').value = upper;
          }
        });

      } catch(err) {
        status.innerText = 'Could not auto-parse clearly. Please fill in fields manually.';
      }
    }

          if(!item.id) item.id = 'st_' + Math.random().toString(36).substr(2, 9);
        });
        
        streets = streets.concat(parsed);
        saveStreetsToStorage(streets);
        closeModal('importModal');
        renderList();
        alert(`Successfully imported ${parsed.length} streets!`);
      } catch(e) {
        alert('Invalid JSON format.');
      }
    }

    function openModal(id) { document.getElementById(id).style.display = 'block'; }
    function closeModal(id) { document.getElementById(id).style.display = 'none'; }

    // Initial Load
    renderList();
  </script>
